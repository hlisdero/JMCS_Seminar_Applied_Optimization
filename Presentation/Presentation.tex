\section{Introduction to IPOPT}

\begin{frame}{IPOPT: Interior Point OPTimizer}

  \begin{itemize}
    \item Paper by Wächter and Biegler in 2005 \cite{wachter2006implementation}.
    \item Implementation originally in Fortran (version 1.x and 2.x), now deprecated
    \item Version 3.x in C++ published in 2006
    \item Since xxxx part of the COIN-OP foundation \cite{coinop_ipopt}.
    \item It is still developed.
  \end{itemize}

\end{frame}

\begin{frame}{IPOPT: Interior Point OPTimizer}


  One of the most popular open-source optimization packages.

  \vfill

  Bindings available in multiple programming languages:

  \begin{itemize}
    \item Python
    \item Rust
    \item Fortran?
  \end{itemize}

  \vfill

  Inspired other packages and subsequent research

\end{frame}
